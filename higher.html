import numpy as np import matplotlib.pyplot as plt import networkx as nx from scipy.integrate import odeint # Setup global modern graphing theme plt.style.use('seaborn-v0_8-whitegrid' if 'seaborn-v0_8-whitegrid' in plt.style.available else 'default') plt.rcParams.update({'font.size': 10, 'axes.labelsize': 11, 'axes.titlesize': 13}) # ===================================================================== # 1. REAL ANALYSIS: \epsilon-N Convergence Visualizer # ===================================================================== def plot_real_analysis(epsilon=0.05): """Plots a_n = 1/n and highlights the N bound where values stay below epsilon.""" n = np.arange(1, 40) a_n = 1 / n N = int(np.ceil(1 / epsilon)) plt.figure(figsize=(6, 4)) plt.plot(n, a_n, 'o-', color='#2563eb', label=r'$a_n = \frac{1}{n}$') plt.axhline(y=epsilon, color='crimson', linestyle='--', label=f'$\epsilon = {epsilon}$') plt.axvline(x=N, color='forestgreen', linestyle=':', label=f'$N = {N}$') plt.fill_between(n, 0, epsilon, color='crimson', alpha=0.1) plt.title("1. Real Analysis: Sequence Convergence Proof") plt.xlabel("Index (n)") plt.ylabel("Value") plt.xlim(1, 35) plt.ylim(-0.05, 1.05) plt.legend() plt.tight_layout() # ===================================================================== # 2. COMPLEX ANALYSIS: Euler's Formula Orbit # ===================================================================== def plot_complex_analysis(degrees=60): """Maps e^{i\theta} on the complex unit circle plane.""" theta_rad = np.radians(degrees) z = np.exp(1j * theta_rad) # Generate unit circle border circle_t = np.linspace(0, 2*np.pi, 200) plt.figure(figsize=(5, 5)) plt.plot(np.cos(circle_t), np.sin(circle_t), color='#cbd5e1', lw=1.5) plt.axhline(0, color='black', lw=0.5); plt.axvline(0, color='black', lw=0.5) # Vector transformation path plt.quiver(0, 0, z.real, z.imag, angles='xy', scale_units='xy', scale=1, color='#2563eb') plt.plot([z.real, z.real], [0, z.imag], 'g--') # imaginary component projection plt.plot([0, z.real], [z.imag, z.imag], 'r--') # real component projection plt.scatter([z.real], [z.imag], color='black', zorder=5) plt.title(f"2. Complex Analysis: $e^{{i \cdot {degrees}^\circ}}$ Transformation") plt.xlabel("Re(z) [cos($\theta$)]") plt.ylabel("Im(z) [sin($\theta$)]") plt.xlim(-1.2, 1.2); plt.ylim(-1.2, 1.2) plt.gca().set_aspect('equal') plt.tight_layout() # ===================================================================== # 3. ABSTRACT ALGEBRA: Modular Multiplicative Group Orbit # ===================================================================== def plot_abstract_algebra(p=11, g=2): """Plots the cyclic group set orbit structure g^k mod p as a discrete step function.""" orbit = [] curr = 1 for k in range(p): orbit.append(curr) curr = (curr * g) % p if curr == 1: break plt.figure(figsize=(6, 4)) plt.stem(range(len(orbit)), orbit, linefmt='#2563eb-', markerfmt='bo', basefmt=' ') plt.title(f"3. Abstract Algebra: Cyclic Group Orbit ($g={g}, p={p}$)") plt.xlabel("Power Sequence exponent index (k)") plt.ylabel(f"$g^k \pmod{{p}}$ Elements") plt.xticks(range(len(orbit))) plt.tight_layout() # ===================================================================== # 4. LINEAR ALGEBRA: 2x2 Transformation Grid & Basis Matrix # ===================================================================== def plot_linear_algebra(matrix_ABCD=[[5, 2], [1, 3]]): """Plots original basis vectors transformed by matrix A to visually show spatial scale alteration.""" A = np.array(matrix_ABCD) i_hat, j_hat = np.array([1, 0]), np.array([0, 1]) # Map transformation vectors i_trans, j_trans = A.dot(i_hat), A.dot(j_hat) det = np.linalg.det(A) plt.figure(figsize=(5, 5)) plt.axhline(0, color='black', lw=0.5); plt.axvline(0, color='black', lw=0.5) # Plot original unit vectors vs transformed vector tracks plt.quiver([0, 0], [0, 0], [i_hat[0], j_hat[0]], [i_hat[1], j_hat[1]], color='#cbd5e1', angles='xy', scale_units='xy', scale=1, label='Basis') plt.quiver([0, 0], [0, 0], [i_trans[0], j_trans[0]], [i_trans[1], j_trans[1]], color=['#2563eb', 'crimson'], angles='xy', scale_units='xy', scale=1, label='Transformed') plt.title(f"4. Linear Algebra\nMatrix Transformation (det = {det:.1f})") plt.xlim(-1, max(i_trans[0], j_trans[0])+2) plt.ylim(-1, max(i_trans[1], j_trans[1])+2) plt.gca().set_aspect('equal') plt.tight_layout() # ===================================================================== # 5. TOPOLOGY: Polyhedron Graph Configuration Invariant # ===================================================================== def plot_topology(): """Generates a planar topological projection layout tracking Euler characteristics.""" # Tetrahedron configuration framework model G = nx.complete_graph(4) plt.figure(figsize=(5, 4)) pos = nx.planar_layout(G) nx.draw(G, pos, with_labels=True, node_color='#2563eb', font_color='white', font_weight='bold', node_size=400, edge_color='#cbd5e1', width=2) plt.title("5. Topology: Tetrahedron Invariant ($\chi = V-E+F = 2$)") plt.tight_layout() # ===================================================================== # 6. NUMBER THEORY & CRYPTOGRAPHY: Modular Exponentiation Step Track # ===================================================================== def plot_number_theory(base=3, power=45, mod=7): """Visualizes the repeated sequence patterns inside a modular domain space wrapper.""" steps = [ (base**i) % mod for i in range(power + 1) ] plt.figure(figsize=(6, 3.5)) plt.plot(steps[:20], 'o-', color='#2563eb', markersize=4) # Display first 20 operations plt.title(f"6. Number Theory: Modulo Residues Step Sequence ($3^k \pmod{{7}}$)") plt.xlabel("Successive Exponent Steps") plt.ylabel("Remainder Value") plt.xticks(range(0, 21, 2)) plt.tight_layout() # ===================================================================== # 7. ORDINARY DIFFERENTIAL EQUATIONS: Exponential Amplitude Tracking # ===================================================================== def plot_ordinary_differential_equations(y0=10, k=-0.5): """Plots explicit solution trajectory over continuous scale runtime domains.""" t = np.linspace(0, 5, 200) y = y0 * np.exp(k * t) plt.figure(figsize=(6, 4)) plt.plot(t, y, lw=2.5, color='#2563eb', label=r'$y(t) = y_0 \cdot e^{kt}$') plt.title(f"7. ODEs: Exponential Decay Trend Flow ($k={k}$)") plt.xlabel("Time Domain Variable (t)") plt.ylabel("System State Level y(t)") plt.legend() plt.tight_layout() # ===================================================================== # 8. PARTIAL DIFFERENTIAL EQUATIONS: Steady-State Temperature Slope # ===================================================================== def plot_partial_differential_equations(u0=80, u1=20): """Plots the linear gradient solution vector for a 1D Steady State Heat Dirichlet environment.""" x = np.linspace(0, 1, 100) u = u0 + x * (u1 - u0) plt.figure(figsize=(6, 4)) plt.plot(x, u, lw=2.5, color='#2563eb') plt.title("8. PDEs: 1D Steady-State Linear Heat Gradient") plt.xlabel("Normalized Boundary Node Coordinate Spatial Profile (x)") plt.ylabel("Temperature u(x)") plt.tight_layout() # ===================================================================== # 9. FUNCTIONAL ANALYSIS: Sequence Element Magnitudes & Space Bound # ===================================================================== def plot_functional_analysis(v=[3, 4, 12]): """Bar visual representing component coordinate sequence inputs against aggregate norm vector thresholds.""" elements = [f"$x_1={v[0]}$", f"$x_2={v[1]}$", f"$x_3={v[2]}$"] squares = [x**2 for x in v] norm = np.sqrt(sum(squares)) plt.figure(figsize=(6, 4)) bars = plt.bar(elements, squares, color='#cbd5e1', edgecolor='gray', width=0.5) bars[2].set_color('#2563eb') # highlight peak vector link plt.title(f"9. Functional Analysis: $l^2$ Norm Grid Profile ($\|x\| = {norm:.1f}$)") plt.ylabel("Squared Value Coordinate Magnitude") plt.tight_layout() # ===================================================================== # 10. GRAPH THEORY: Complete Symmetric K_n Graph Circle # ===================================================================== def plot_graph_theory(num_nodes=5): """Generates the circular network map matrix specified in the HTML Canvas logic.""" G = nx.complete_graph(num_nodes) pos = nx.circular_layout(G) plt.figure(figsize=(5, 5)) nx.draw(G, pos, with_labels=True, node_color='#2563eb', font_color='white', font_weight='bold', node_size=450, edge_color='#cbd5e1') plt.title(f"10. Graph Theory: Complete Symmetric Matrix $K_{num_nodes}$") plt.tight_layout() # ===================================================================== # 11. OPERATOR THEORY: Resolvent Mapping Spectrum Evaluation Range # ===================================================================== def plot_operator_theory(k=2): """Evaluates and tracks the linear scaling transform metrics across standard variable ranges.""" lam = np.linspace(k+0.1, k+5, 200) resolvent_val = 1 / (lam - k) plt.figure(figsize=(6, 4)) plt.plot(lam, resolvent_val, lw=2.5, color='#2563eb', label=r'$R(\lambda) = (\lambda I - T)^{-1}$') plt.axvline(x=k, color='crimson', linestyle='--', label='Spectrum Boundary Pool (Singular Point)') plt.title("11. Operator Theory: Resolvent Spectrum Inversion Mapping") plt.xlabel("Eigenvalue Spectrum Parameter ($\lambda$)") plt.ylabel("Resolvent Operator Intensity Spectrum") plt.legend() plt.tight_layout() # ===================================================================== # 12. BANACH ALGEBRA: Norm Sub-multiplicativity verification boundary # ===================================================================== def plot_banach_algebra(norm_A=4, norm_B=3): """Draws inequality check spaces visually bounding total multiplication envelope limits.""" max_bound = norm_A * norm_B categories = ['||A||', '||B||', '||A|| * ||B||\n(Max Sub-multiplicative Bound)'] values = [norm_A, norm_B, max_bound] plt.figure(figsize=(6, 4)) plt.bar(categories, values, color=['#cbd5e1', '#cbd5e1', '#2563eb'], width=0.4) plt.title("12. Banach Algebra: Algebra Norm Metric Properties") plt.ylabel("Upper Norm Metric Value Threshold") plt.tight_layout() # ===================================================================== # 13. MATHEMATICAL METHODS: Fourier Square Wave Harmonic Target Expansion # ===================================================================== def plot_mathematical_methods(harmonics_cutoff=5): """Plots a square wave reconstruction approximation via partial Fourier summation steps.""" t = np.linspace(-np.pi, np.pi, 500) f = np.zeros_like(t) for n in range(1, harmonics_cutoff + 1, 2): f += (4 / np.pi) * (np.sin(n * t) / n) plt.figure(figsize=(6, 4)) plt.plot(t, f, lw=2, color='#2563eb', label=f'{harmonics_cutoff} Harmonics') # Perfect base trace template loop plt.plot(t, np.sign(np.sin(t)), 'k--', alpha=0.5, label='Ideal Target Square Wave') plt.title("13. Mathematical Methods: Fourier Series Harmonics Approximation") plt.xlabel("Time Cycle Scale Phase Step Parameter (t)") plt.ylabel("Wave Amplitude Velocity f(t)") plt.legend(loc='lower right') plt.tight_layout() # ===================================================================== # 14. RELATIVITY: Relativistic Lorentz Time Dilation Curve # ===================================================================== def plot_relativity(stationary_time=10): """Plots the exponential upward bending curve scaling with operational shift factors.""" vc_ratio = np.linspace(0, 0.95, 200) dilated_time = stationary_time / np.sqrt(1 - vc_ratio**2) plt.figure(figsize=(6, 4)) plt.plot(vc_ratio, dilated_time, lw=2.5, color='#2563eb') plt.axvline(x=1.0, color='crimson', linestyle=':', label='Asymptote at light speed ($c$)') plt.title(r"14. Relativity: Lorentz Time Dilation Curve ($\Delta t'$)") plt.xlabel("Fractional Speed Parameter Velocity ($v/c$)") plt.ylabel("Observed Dilated Time Scale (seconds)") plt.legend() plt.tight_layout() # ===================================================================== # 15. CLASSICAL MECHANICS: Pendulum Period Variation Across Length Scales # ===================================================================== def plot_classical_mechanics(g=9.81): """Plots simple pendulum period function bounds scaling across length coordinates.""" lengths = np.linspace(0.1, 5, 200) periods = 2 * np.pi * np.sqrt(lengths / g) plt.figure(figsize=(6, 4)) plt.plot(lengths, periods, lw=2.5, color='#2563eb') plt.title("15. Classical Mechanics: Simple Pendulum Period Scale Tracker") plt.xlabel("Cord String Length Parameter Line (meters)") plt.ylabel("Total Swings Angular Cycle Period (seconds)") plt.tight_layout() # ===================================================================== # 16. DIFFERENTIAL GEOMETRY: 3D Geometric Space Helix Mapping & Tangent # ===================================================================== def plot_differential_geometry(t_target=3.0): """Maps continuous three dimensional spiral curve grids accompanied by instantaneous velocity directions.""" fig = plt.figure(figsize=(6, 5)) ax = fig.add_subplot(111, projection='3d') t = np.linspace(0, 12, 300) x, y, z = np.cos(t), np.sin(t), 0.2 * t ax.plot(x, y, z, color='#cbd5e1', lw=2, label='Space Curve path $r(t)$') # Target calculations matching your HTML slide configurations x_pt, y_pt, z_pt = np.cos(t_target), np.sin(t_target), 0.2 * t_target dx, dy, dz = -np.sin(t_target), np.cos(t_target), 0.2 ax.scatter([x_pt], [y_pt], [z_pt], color='crimson', s=40, zorder=10) ax.quiver(x_pt, y_pt, z_pt, dx, dy, dz, length=1.2, color='#2563eb', lw=2, label="Tangent Vector track $r'(t)$") ax.set_title("16. Differential Geometry: Helix Intrinsic Surface Curve Mapping") ax.set_xlabel('X'); ax.set_ylabel('Y'); ax.set_zlabel('Z') ax.legend(loc='upper left') plt.tight_layout() # ===================================================================== # GLOBAL GRAPH RENDER INTERFACE SELECTOR # ===================================================================== if __name__ == "__main__": # Toggle any plot function by removing the '#' comment markers below: plot_real_analysis() plot_complex_analysis() plot_abstract_algebra() plot_linear_algebra() plot_topology() plot_number_theory() plot_ordinary_differential_equations() plot_partial_differential_equations() plot_functional_analysis() plot_graph_theory() plot_operator_theory() plot_banach_algebra() plot_mathematical_methods() plot_relativity() plot_classical_mechanics() plot_differential_geometry() # Display active selections plt.show()
